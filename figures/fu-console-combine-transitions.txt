(let* ((img1 (car (gimp-file-load RUN-NONINTERACTIVE "/tikhome/jscheunemann/hauptseminar-report/figures/classical_transition_a1_c_a3.png" "first.png")))
       (img2 (car (gimp-file-load RUN-NONINTERACTIVE "/tikhome/jscheunemann/hauptseminar-report/figures/classical_transition_a1_a2_b2.png" "second.png")))
       (w1 (car (gimp-image-width img1)))
       (h1 (car (gimp-image-height img1)))
       (w2 (car (gimp-image-width img2)))
       (h2 (car (gimp-image-height img2)))
       (max-h (max h1 h2))
       (new-img (car (gimp-image-new (+ w1 w2) max-h RGB)))
       (layer1 (car (gimp-file-load-layer RUN-NONINTERACTIVE new-img "/tikhome/jscheunemann/hauptseminar-report/figures/classical_transition_a1_c_a3.png")))
       (layer2 (car (gimp-file-load-layer RUN-NONINTERACTIVE new-img "/tikhome/jscheunemann/hauptseminar-report/figures/classical_transition_a1_a2_b2.png")))
       (bg-layer (car (gimp-layer-new new-img (+ w1 w2) max-h RGB-IMAGE "background" 100 LAYER-MODE-NORMAL))))
  ; add a white background layer first so there's no transparency/black gaps
  (gimp-image-insert-layer new-img bg-layer 0 -1)
  (gimp-image-set-active-layer new-img bg-layer)
  (gimp-context-set-foreground '(255 255 255))
  (gimp-image-select-rectangle new-img CHANNEL-OP-REPLACE 0 0 (+ w1 w2) max-h)
  (gimp-edit-fill bg-layer FILL-FOREGROUND)
  (gimp-selection-none new-img)
  ; insert and vertically center layer1
  (gimp-image-insert-layer new-img layer1 0 -1)
  (gimp-layer-set-offsets layer1 0 (/ (- max-h h1) 2))
  ; insert and vertically center layer2
  (gimp-image-insert-layer new-img layer2 0 -1)
  (gimp-layer-set-offsets layer2 w1 (/ (- max-h h2) 2))
  (gimp-image-flatten new-img)
  (file-png-save RUN-NONINTERACTIVE new-img (car (gimp-image-get-active-drawable new-img)) "/tikhome/jscheunemann/hauptseminar-report/figures/classical_transitions_combined.png" "output" 0 9 1 1 1 1 1))
